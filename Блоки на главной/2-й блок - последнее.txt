<script>
document.addEventListener("DOMContentLoaded", function() {
    const grid = document.getElementById('latestGrid');
    const prev = document.getElementById('latestBtnPrev');
    const next = document.getElementById('latestBtnNext');
    const fadeLeft = document.getElementById('fadeLeft');
    const fadeRight = document.getElementById('fadeRight');

    if (!grid) return;

    function checkLatestScroll() {
        const isStart = grid.scrollLeft <= 20;
        const isEnd = Math.ceil(grid.scrollLeft) >= (grid.scrollWidth - grid.clientWidth - 20);

        if (isStart) {
            prev.classList.remove('visible');
            fadeLeft.style.opacity = '0';
        } else {
            prev.classList.add('visible');
            fadeLeft.style.opacity = '1';
        }

        if (isEnd) {
            next.classList.remove('visible');
            fadeRight.classList.add('hidden');
        } else {
            next.classList.add('visible');
            fadeRight.classList.remove('hidden');
        }
    }

    function getScrollStep() {
        const card = grid.querySelector('.latest-card');
        return card ? card.offsetWidth + 24 : 400;
    }

    if (next) {
        next.addEventListener('click', function() {
            grid.scrollBy({ left: getScrollStep(), behavior: 'smooth' });
        });
    }

    if (prev) {
        prev.addEventListener('click', function() {
            grid.scrollBy({ left: -getScrollStep(), behavior: 'smooth' });
        });
    }

    grid.addEventListener('scroll', checkLatestScroll);
    window.addEventListener('resize', checkLatestScroll);
    setTimeout(checkLatestScroll, 100);
});
</script>
