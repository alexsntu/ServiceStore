<script>
document.addEventListener("DOMContentLoaded", function() {
    const grid = document.getElementById('accessGrid');
    const prev = document.getElementById('accessBtnPrev');
    const next = document.getElementById('accessBtnNext');

    if (!grid) return;

    function checkAccessScroll() {
        const isStart = grid.scrollLeft <= 20;
        const isEnd = Math.ceil(grid.scrollLeft) >= (grid.scrollWidth - grid.clientWidth - 20);

        if (isStart) {
            prev.classList.remove('visible');
        } else {
            prev.classList.add('visible');
        }

        if (isEnd) {
            next.classList.remove('visible');
        } else {
            next.classList.add('visible');
        }
    }

    function getScrollStep() {
        const productCard = grid.querySelector('.product-type');
        return productCard ? productCard.offsetWidth + 24 : 339;
    }

    if (next) {
        next.addEventListener('click', function() {
            grid.scrollBy({ left: getScrollStep(), behavior: 'smooth' });
        });
    }

    if (prev) {
        prev.addEventListener('click', function() {
            grid.scrollBy({ left: -getScrollStep(), behavior: 'smooth' });
        });
    }

    grid.addEventListener('scroll', checkAccessScroll);
    window.addEventListener('resize', checkAccessScroll);
    setTimeout(checkAccessScroll, 100);
});
</script>
