<script>
document.addEventListener("DOMContentLoaded", function() {
    const navGrid = document.getElementById('navGrid');
    const btnPrev = document.getElementById('navBtnPrev');
    const btnNext = document.getElementById('navBtnNext');

    if (!navGrid) return;

    function checkNavScroll() {
        if (navGrid.scrollLeft > 10) {
            btnPrev.classList.add('visible');
        } else {
            btnPrev.classList.remove('visible');
        }

        if (Math.ceil(navGrid.scrollLeft) >= (navGrid.scrollWidth - navGrid.clientWidth - 10)) {
            btnNext.classList.remove('visible');
        } else {
            if (navGrid.scrollWidth > navGrid.clientWidth) {
                btnNext.classList.add('visible');
            } else {
                btnNext.classList.remove('visible');
            }
        }
    }

    if (btnNext) {
        btnNext.addEventListener('click', function() {
            navGrid.scrollBy({ left: 150, behavior: 'smooth' });
        });
    }

    if (btnPrev) {
        btnPrev.addEventListener('click', function() {
            navGrid.scrollBy({ left: -150, behavior: 'smooth' });
        });
    }

    navGrid.addEventListener('scroll', checkNavScroll);
    window.addEventListener('resize', checkNavScroll);
    setTimeout(checkNavScroll, 100);
});
</script>
